## 10. Independent oracle (source)

Run as `node oracle.cjs <this file> dist/_redirects`. It reads the fenced matrix in section 2 and imports nothing from the repository, so it shares no logic with `route-history.ts`.

```js
// usage: node oracle.cjs <evidence.md> <dist/_redirects>
// Independent test oracle (T19.0): compares a generated _redirects file to the frozen matrix in the evidence file.
// Deliberately imports nothing from the repo (no route-history.ts).
const fs = require('fs');
const [evidence, redirects] = process.argv.slice(2);
const md = fs.readFileSync(evidence, 'utf8');
const block = /```text\r?\n([\s\S]*?)```/.exec(md.slice(md.indexOf('Independently frozen expected matrix')));
if (!block) { console.error('ORACLE: frozen matrix block not found'); process.exit(2); }
const expected = block[1].split(/\r?\n/).filter(Boolean).map((l) => { const [a, b] = l.split(' -> '); return `${a} ${b} 301`; }).sort();
const actual = fs.readFileSync(redirects, 'utf8').split(/\r?\n/).filter(Boolean).sort();
const missing = expected.filter((l) => !actual.includes(l));
const extra = actual.filter((l) => !expected.includes(l));
console.log(`ORACLE: frozen=${expected.length} generated=${actual.length} missing=${missing.length} extra=${extra.length}`);
for (const l of missing) console.log('  MISSING  ' + l);
for (const l of extra) console.log('  EXTRA    ' + l);
process.exit(missing.length || extra.length ? 1 : 0);
```
